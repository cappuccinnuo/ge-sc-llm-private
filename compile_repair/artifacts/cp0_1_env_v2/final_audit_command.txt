import ast, datetime, hashlib, json, os, subprocess
from pathlib import Path
from scripts.DFG.dfg_v2.experiment.run import _code_refs
root=Path.cwd();out=root/'compile_repair/artifacts/cp0_1_env_v2'
baseline=json.loads((out/'baseline.json').read_text())
def sha(path):return hashlib.sha256(Path(path).read_bytes()).hexdigest()
checks={};differences={}
for group in ('frozen_tracked','old_artifacts','fixture_sha256','mando_metadata','production_metadata','solc_files'):
    refs=baseline[group];bad=[]
    for name,expected in refs.items():
        path=Path(name) if Path(name).is_absolute() else root/name
        if not path.is_file() or sha(path)!=expected:bad.append(name)
    checks[group+'_unchanged']=not bad;differences[group]=bad
    print(group,len(refs),'differences',len(bad),flush=True)
current_solc={str(p):sha(p) for p in (Path.home()/'.solc-select').rglob('*') if p.is_file()}
checks['no_solc_file_added_or_removed']=current_solc==baseline['solc_files']
checks['SOLC_VERSION_unchanged']=os.environ.get('SOLC_VERSION')==baseline['SOLC_VERSION']
refs=_code_refs();checks['code_refs_354_unchanged']=refs==baseline['code_refs'] and len(refs)==354
for name in ('CP0_1_REPORT.md','PROGRESS.md','docs/CHANGE_LOG.md'):
    path='compile_repair/'+name;ref=baseline['documents'][path]
    checks[name+'_historical_prefix_unchanged']=hashlib.sha256((root/path).read_bytes()[:ref['bytes']]).hexdigest()==ref['sha256']
for name,marker in [('docs/HANDOVER.md','## 6. 交接记录'),('docs/DECISIONS.md','## 2. r1 的决策编号')]:
    path='compile_repair/'+name
    previous=subprocess.check_output(['git','show',baseline['head']+':'+path]).decode()
    old=previous[previous.index(marker):];current=(root/path).read_text()
    checks[name+'_history_preserved']=current[current.index(marker):].startswith(old)
changes=subprocess.check_output(['git','diff','--name-only',baseline['head']]).decode().splitlines()
checks['only_compile_repair_tracked_changes']=all(p.startswith('compile_repair/') for p in changes)
production=sorted(str(p.relative_to(root)) for p in (root/'compile_repair').rglob('*.py') if not any(x in p.parts for x in ('artifacts','tests')))
checks['production_file_count_9']=len(production)==9
checks['production_files_python37_parse']=all(ast.parse((root/p).read_text()) for p in production)
allowed={'compile_repair/core/model.py','compile_repair/toolchain/solc.py','compile_repair/toolchain/slither.py'}
checks['production_edits_only_three']=set(p for p in changes if p in production)==allowed
checks['prompts_unchanged']=(root/'compile_repair/PROMPTS.md').read_bytes()==subprocess.check_output(['git','show',baseline['head']+':compile_repair/PROMPTS.md'])
summary={'status':'passed' if all(checks.values()) else 'failed','checks':checks,'differences':differences,
    'reference_counts':{k:len(baseline[k]) for k in ('frozen_tracked','old_artifacts','fixture_sha256','mando_metadata','production_metadata','solc_files')},
    'code_refs_count':len(refs),'tracked_changes':changes,'production_files':production,
    'finished_utc':datetime.datetime.now(datetime.timezone.utc).isoformat()}
with (out/'frozen_audit.json').open('x',encoding='utf8') as f:json.dump(summary,f,indent=2,ensure_ascii=False);f.write('\n')
print(json.dumps({'status':summary['status'],'checks':checks},indent=2),flush=True)
assert all(checks.values()),'Stop: frozen-file audit differs'
