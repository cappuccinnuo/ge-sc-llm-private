import datetime, hashlib, json, os, subprocess, sys, time
from pathlib import Path
root=Path.cwd()
out=root/'compile_repair/artifacts/cp0_1_env_v2'
name=sys.argv[1]
envs={'production':'/home/zhao/miniconda3/envs/compile-repair-production/bin/python',
      'mando':'/home/zhao/miniconda3/envs/mando/bin/python'}
profile='production' if 'production' in name else 'mando'
targets=(['compile_repair/tests/test_settings.py','compile_repair/tests/test_layering.py']
         if name.startswith('unit_') else ['scripts/DFG/dfg_v2/tests'] if name=='pytest_dfg'
         else ['compile_repair/tests'])
cmd=[envs[profile],'-B','-m','pytest','-q','-ra','-x','-p','no:cacheprovider']+targets+['--basetemp='+str(out/(name+'_tmp'))]
assert not (out/(name+'_tmp')).exists()
env=dict(os.environ,PYTHONDONTWRITEBYTECODE='1',OMP_NUM_THREADS='2',MKL_NUM_THREADS='2',
         TREE_SITTER_LIB_DIR='/home/zhao/SmartContract/ge-sc-llm/build/linux-x86_64')
if profile=='production': env['PYTHONNOUSERSITE']='1'
start=time.monotonic()
with (out/(name+'.log')).open('x',encoding='utf8') as log:
    result=subprocess.run(cmd,cwd=str(root),env=env,stdout=log,stderr=subprocess.STDOUT)
record={'command':cmd,'cwd':str(root),'profile':profile,'returncode':result.returncode,
        'elapsed_seconds':time.monotonic()-start,'finished_utc':datetime.datetime.now(datetime.timezone.utc).isoformat(),
        'environment':{k:env.get(k) for k in ['PYTHONDONTWRITEBYTECODE','PYTHONNOUSERSITE','OMP_NUM_THREADS','MKL_NUM_THREADS','TREE_SITTER_LIB_DIR']},
        'log_sha256':hashlib.sha256((out/(name+'.log')).read_bytes()).hexdigest(),'cost_usd':0,'automatic_retry':False}
with (out/(name+'.json')).open('x',encoding='utf8') as f:json.dump(record,f,indent=2);f.write('\n')
print(json.dumps(record,indent=2));print((out/(name+'.log')).read_text()[-5000:])
sys.exit(result.returncode)
