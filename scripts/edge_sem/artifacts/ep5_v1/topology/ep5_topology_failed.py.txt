"""EP5 CPU topology diagnostics, with exact original node identities.

Evidence sets use original-view frozen facts and sealed GUARDS sources.
Reachability counts distinct other nodes at directed shortest distance 1 or 2
in the actual reflected input; it is not a prediction or vulnerability score.
"""
import copy
import csv
from collections import Counter,defaultdict
import networkx as nx
import numpy as np
from sco_models import graph_utils as gu
from scripts.DFG.dfg_v2.model_input import edge_data
from scripts.DFG.dfg_v2.benchmark_v2.feature_view import graph_fingerprint
from scripts.node_sem.carrier import row_key
from scripts.node_sem.facts_io import load_context
from scripts.node_sem.facts import extract
from scripts.edge_sem.materialize import materialize
from scripts.edge_sem.placebo import randomize_edges,permute_attributes,degree_signature,DEFAULT_SEED
from scripts.edge_sem.build_f4 import load_edges
from scripts.edge_sem.attr_batch import load_attributes,facts_digest,numeric_digest
from scripts.edge_sem.attrs import edge_rows
from .common import ROOT,read,write,checked,ref
from .ep5_common import OUT,TASKS

CASES=ROOT/"scripts/node_sem/artifacts/summary_NS_v1/cases_attempt2/persistent_cases.json"
CATEGORIES=("external_call","state_write","guard_condition","modifier_definition")


def reflected_graph(graph,profile,additions=()):
    """Use the production edge adapter, then restore exact original identities."""
    work=copy.deepcopy(graph)
    for i,e in enumerate(additions):
        s,t=e["src"],e["dst"]
        if any(n not in work or work.nodes[n]["source_file"]!=e["file"] for n in (s,t)):
            raise ValueError("New edge identity/cross-file mismatch")
        key=("ep5_topology",i)
        if work.has_edge(s,t,key):raise ValueError("Topology edge key collision")
        work.add_edge(s,t,key=key,edge_type=e["relation"])
    gu.add_hetero_ids(work)
    inverse=defaultdict(list)
    for n,a in work.nodes(data=True):
        if a["node_hetero_id"]!=len(inverse[a["node_type"]]):
            raise AssertionError("Frozen type-local row differs")
        inverse[a["node_type"]].append(n)
    data,meta=materialize(edge_data(work),profile)
    out=nx.MultiDiGraph()
    out.add_nodes_from((n,dict(node_type=a["node_type"],source_file=a["source_file"])) for n,a in graph.nodes(data=True))
    for (s,r,t),(u,v) in data.items():
        for a,b in zip(u.tolist(),v.tolist()):
            out.add_edge(inverse[s][a],inverse[t][b],edge_type=r)
    if list(out)!=list(graph):raise AssertionError("Node identity/order differs")
    return out,data


def node_degrees(graph):
    incoming=Counter();outgoing=Counter()
    for s,t,a in graph.edges(data=True):
        c=(graph.nodes[s]["node_type"],a["edge_type"],graph.nodes[t]["node_type"])
        outgoing[(s,c)]+=1;incoming[(t,c)]+=1
    return incoming,outgoing


def graph_statistics(graph,data):
    groups=defaultdict(list)
    for n,a in graph.nodes(data=True):groups[a["node_type"]].append(n)
    return dict(nodes=graph.number_of_nodes(),edges=graph.number_of_edges(),
        canonical_relations=len(data),relation_names=len({c[1] for c in data}),
        by_relation=dict(sorted(Counter(a["edge_type"] for _,_,a in graph.edges(data=True)).items())),
        by_canonical=[dict(canonical=list(c),edges=int(len(data[c][0]))) for c in sorted(data)],
        indegree_by_node_type={typ:dict(nodes=len(ns),incoming_edges=sum(graph.in_degree(n) for n in ns),
            mean_indegree=sum(graph.in_degree(n) for n in ns)/len(ns),zero_indegree=sum(graph.in_degree(n)==0 for n in ns))
            for typ,ns in sorted(groups.items())})


def evidence_nodes(graph,facts,edges):
    kinds={k:set() for k in CATEGORIES}
    for n,a in graph.nodes(data=True):
        f=facts[row_key(n,a)]
        if f["node_kind"] in ("statement","condition"):
            if f["external"].get("kinds"):kinds["external_call"].add(n)
            if f["state"].get("writes"):kinds["state_write"].add(n)
        if f["node_kind"]=="modifier":kinds["modifier_definition"].add(n)
    kinds["guard_condition"]={e["src"] for e in edges if e["relation"]=="GUARDS"}
    return kinds


def reachable_evidence(graph,start,evidence):
    distances=nx.single_source_shortest_path_length(graph,start,cutoff=2)
    distances.pop(start,None) # no zero-hop self evidence, even on a cycle
    reached=set(distances);union=set().union(*evidence.values())
    return dict(counts={k:len(reached&v) for k,v in evidence.items()},
        total_unique=len(reached&union),
        evidence={k:sorted(reached&v,key=str) for k,v in evidence.items()},
        distance=[dict(node=n,hops=distances[n]) for n in sorted(reached&union,key=str)])


def validate_case(case,task,context,graph):
    n=case["original_graph_node_id"];file=case["file"]
    if case["task_id"]!=task or case["case_type"]!="persistent_fn" or case["all_missed"] is not True:
        raise ValueError("Unexpected persistent case identity")
    if n not in graph or graph.nodes[n]["source_file"]!=file or graph.nodes[n]["node_type"]!=case["node_type"]:
        raise ValueError("Case graph node/file/type identity differs")
    source=context["sources"][file]
    if source["sha256"]!=case["source_sha256"] or source["sha256"]!=case["source"]["sha256"]:
        raise ValueError("Case source SHA differs")
    checked(case["source"]);checked(case["graph"])
    if case["graph"]["sha256"]!=context["provenance"]["graphs"]["F3"]["sha256"]:
        raise ValueError("Case/frozen graph reference differs")
    if case["graph"]["fingerprint"]!=graph_fingerprint(graph):
        raise ValueError("Case graph fingerprint differs")


def main():
    if not read(OUT/"attribute_delta.json")["passed"]:raise ValueError("Section 0 gate not passed")
    dest=OUT/"topology";dest.mkdir(exist_ok=False)
    implementation=[ref(ROOT/"scripts/edge_sem"/p) for p in ("recon/ep5_topology.py","placebo.py","materialize.py")]
    write(dest/"started.json",dict(cases=ref(CASES),implementation=implementation,seed=DEFAULT_SEED,
        categories=dict(external_call="statement/condition with a confirmed external.kinds fact",
            state_write="statement/condition with a confirmed state.writes fact",
            guard_condition="source node of sealed GUARDS evidence",modifier_definition="frozen node_kind modifier"),
        reachability="directed shortest path 1..2 hops, distinct nodes, start excluded, union deduplicated",
        training_executed=False,gpu_executed=False))
    cases=[c for c in read(CASES) if c["case_type"]=="persistent_fn" and c["task_id"] in TASKS]
    identities=[(c["task_id"],c["original_graph_node_id"]) for c in cases]
    if len(identities)!=len(set(identities)):raise ValueError("Duplicate persistent node case")
    result={};case_rows=[]
    for task in TASKS:
        print(task,"topology/facts loading",flush=True)
        ctx=load_context(task);g=ctx["graphs"]["F3"];before=graph_fingerprint(g)
        path=OUT/"attributes"/task/"manifest.json";manifest=read(path)
        edges=load_edges(checked(manifest["f4_manifest"]))
        attrs=load_attributes(path,g,edges);rows=edge_rows(g,edges);keys=[r["key"] for r in rows]
        facts=extract(task,context=ctx)
        if facts_digest(facts)!=read(checked(manifest["facts"]))["facts_sha256"]:
            raise ValueError("Current facts differ from EP5 attribute provenance")
        evidence=evidence_nodes(g,facts,edges)
        taskdir=dest/task;taskdir.mkdir()
        write(taskdir/"evidence_nodes.json",dict(categories={k:sorted(v,key=str) for k,v in evidence.items()},
            facts=manifest["facts"],source_refs=ctx["sources"]))
        randomized,random_audit=randomize_edges(g,edges)
        if degree_signature(g,randomized)!=degree_signature(g,edges):raise AssertionError("E1 degrees differ")
        write(taskdir/"placebo_edges.json",randomized)
        write(taskdir/"placebo_edge_audit.json",random_audit)
        graphs={};stats={}
        for profile,extra in (("G3R",()),("G3R-T",()),("G4R",edges),("G4R-P",randomized)):
            h,data=reflected_graph(g,"G4R" if profile=="G4R-P" else profile,extra)
            graphs[profile]=h;stats[profile]=graph_statistics(h,data)
        if node_degrees(graphs["G4R"])!=node_degrees(graphs["G4R-P"]):
            raise AssertionError("Reflected placebo canonical degrees differ")
        write(taskdir/"graph_statistics.json",stats)
        permuted,permutation=permute_attributes(rows,attrs)
        matrix=np.stack([permuted[k] for k in keys])
        with (taskdir/"permuted_attrs.npz").open("xb") as f:
            np.savez_compressed(f,edge_keys=np.asarray(keys),attrs=matrix)
        write(taskdir/"permutation_audit.json",permutation)
        write(taskdir/"permutation_manifest.json",dict(source=ref(path),seed=DEFAULT_SEED,
            output=ref(taskdir/"permuted_attrs.npz"),audit=ref(taskdir/"permutation_audit.json"),
            numeric_digest=numeric_digest(keys,matrix),
            note="Forward vectors permuted as a whole within relation; reverse vectors must copy the corresponding permuted forward row."))
        selected=[c for c in cases if c["task_id"]==task]
        for c in selected:
            validate_case(c,task,ctx,g);n=c["original_graph_node_id"]
            row={k:c[k] for k in ("task_id","file","source_sha256","source_id","original_graph_node_id","node_type","fold","checkpoint")}
            row["profiles"]={p:reachable_evidence(graphs[p],n,evidence) for p in ("G3R","G3R-T","G4R")}
            row["delta_G4R_minus_G3R"]={k:row["profiles"]["G4R"]["counts"][k]-row["profiles"]["G3R"]["counts"][k] for k in CATEGORIES}
            row["delta_total_unique"]=row["profiles"]["G4R"]["total_unique"]-row["profiles"]["G3R"]["total_unique"]
            row["added_evidence"]={k:sorted(set(row["profiles"]["G4R"]["evidence"][k])-set(row["profiles"]["G3R"]["evidence"][k]),key=str) for k in CATEGORIES}
            case_rows.append(row)
        if graph_fingerprint(g)!=before:raise AssertionError("Original graph mutated")
        result[task]=dict(graph_statistics=ref(taskdir/"graph_statistics.json"),attributes=ref(path),
            evidence=ref(taskdir/"evidence_nodes.json"),placebo_audit=ref(taskdir/"placebo_edge_audit.json"),
            permutation=ref(taskdir/"permutation_manifest.json"),cases=len(selected),
            increased_cases=sum(r["delta_total_unique"]>0 for r in case_rows if r["task_id"]==task),
            evidence_counts={k:len(v) for k,v in evidence.items()},
            original_graph_unchanged=True,reflected_placebo_degrees_exact=True)
        print(task,"cases",len(selected),"increased",result[task]["increased_cases"],flush=True)
    write(dest/"reachability.json",case_rows)
    csvrows=[]
    for row in case_rows:
        flat={k:v for k,v in row.items() if k not in ("profiles","delta_G4R_minus_G3R","added_evidence")}
        for profile,values in row["profiles"].items():
            flat.update({profile+"_"+k:v for k,v in values["counts"].items()})
            flat[profile+"_total_unique"]=values["total_unique"]
        csvrows.append(flat)
    with (dest/"reachability.csv").open("x",encoding="utf8",newline="") as stream:
        writer=csv.DictWriter(stream,fieldnames=list(csvrows[0]));writer.writeheader();writer.writerows(csvrows)
    write(dest/"summary.json",dict(passed=True,tasks=result,cases=ref(CASES),
        reachability=ref(dest/"reachability.json"),table=ref(dest/"reachability.csv"),
        any_case_gains_evidence=any(r["delta_total_unique"]>0 for r in case_rows),
        total_cases=len(case_rows),implementation=implementation,training_executed=False,gpu_executed=False,
        anonymity_gate_independent=True))
if __name__=="__main__":main()
