from pathlib import Path
import json,hashlib,math,csv,statistics,collections,datetime,subprocess
from scripts.edge_sem.recon.ep7_original import frozen
R=Path("scripts/edge_sem/artifacts/ep7_original_v1").resolve()
D=R/"completion";D.mkdir(exist_ok=False)
def read(p): return json.loads(Path(p).read_text(encoding="utf8"))
cache={}
def ref(p):
    p=Path(p).resolve()
    if p not in cache:
        h=hashlib.sha256()
        with p.open("rb") as f:
            for chunk in iter(lambda:f.read(4*1024*1024),b""):h.update(chunk)
        cache[p]={"path":str(p),"sha256":h.hexdigest(),"bytes":p.stat().st_size}
    return cache[p]
def check(r):
    actual=ref(r["path"]);assert actual["sha256"]==r["sha256"],r["path"]
    if "bytes" in r: assert actual["bytes"]==r["bytes"],r["path"]
    return Path(r["path"])
def write(p,x):
    with Path(p).open("x",encoding="utf8") as f:json.dump(x,f,ensure_ascii=False,sort_keys=True,indent=2);f.write("\n")
def close(a,b): assert math.isclose(float(a),float(b),rel_tol=1e-12,abs_tol=1e-9),(a,b)
def metric(rows,ck):
    cm=[0,0,0,0]
    for x in rows:
        y=x["target"];p=x[ck]
        assert y in (0,1) and p in (0,1)
        v=x[ck+"_positive_probability"];assert math.isfinite(v) and 0<=v<=1
        cm[y*2+p]+=1
    tn,fp,fn,tp=cm
    div=lambda a,b:100*a/b if b else 0.
    f1=div(2*tp,2*tp+fp+fn);f0=div(2*tn,2*tn+fp+fn)
    return dict(buggy_f1=f1,macro_f1=(f0+f1)/2,precision=div(tp,tp+fp),recall=div(tp,tp+fn),tn_fp_fn_tp=cm,n=len(rows),positive=tp+fn)
def csvwrite(p,rows):
    with Path(p).open("x",encoding="utf8",newline="") as f:
        w=csv.DictWriter(f,fieldnames=list(rows[0]));w.writeheader();w.writerows(rows)

                minimum_free_mib=q["minimum_free_mib"],memory_tight=q["memory_tight"],possible_memory_spill=q["possible_memory_spill"]))
