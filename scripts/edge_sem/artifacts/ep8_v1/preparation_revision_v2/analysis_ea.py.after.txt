PROFILES = ("G3R", "G3R-T", "G4R", "G4R-P")
CONTRASTS = (("G4R", "G3R-T"), ("G4R", "G4R-P"), ("G3R-T", "G3R"))
CHECKPOINTS = ("best", "last")



def preparation_index(root):
    """Resolve the accepted attempt without rewriting an earlier preparation."""
    root=Path(root)
    pointer=root/"preparation_index.json"
    if read(pointer.with_suffix(".seal.json")).get("pointer")!=ref(pointer):
        raise ValueError("Preparation pointer seal differs")
    path=checked(read(pointer)["index"])
    if root.resolve() not in path.resolve().parents:
        raise ValueError("Preparation index escapes this batch")
    if read(path.with_suffix(".seal.json")).get("index")!=ref(path):
        raise ValueError("Preparation index seal differs")
    return path,read(path)


def csv_table(path, rows):
    rows = list(rows)
    columns = list(dict.fromkeys(k for row in rows for k in row))
    with Path(path).open("x", encoding="utf8", newline="") as stream:
        writer = csv.DictWriter(stream, fieldnames=columns, lineterminator="\n")
        writer.writeheader()
        for row in rows:
            writer.writerow({k: json.dumps(v, ensure_ascii=False, sort_keys=True)
                             if isinstance(v, (dict, list, tuple)) else v for k, v in row.items()})


def score(rows, checkpoint="best", stratum="all"):
    # With node_views=None the frozen scorer retains row.target (L-audit).
    # Its stratum test is the exact original 8C filename/cohort definition.
    values = [dict(r, best=r[checkpoint]) for r in rows]
    result = scoring.metric_counts(scoring.confusion(
        scoring.scored_pairs(values, None, "orig", stratum)))
    result.update(precision_wilson_95=scoring.wilson(result["tp"], result["tp"] + result["fp"]),
                  recall_wilson_95=scoring.wilson(result["tp"], result["tp"] + result["fn"]),
                  errors=result["fp"] + result["fn"])
    return result


def relative_error_reduction(base_errors, current_errors):
    # A perfect baseline has no defined relative error reduction.
    return None if base_errors == 0 else 100.0 * (base_errors - current_errors) / base_errors


def align_predictions(rows, labels, expected_keys=None):
    table = scoring.index_rows(rows)
    if expected_keys is not None and set(table) != set(expected_keys):
        raise ValueError("Prediction population differs")
    for key, row in table.items():
        if key not in labels:
            raise ValueError("Prediction has no frozen node identity")
        label = labels[key]
        if not label["L-audit_include"] or row["target"] != label["L-audit_target"]:
            raise ValueError("Prediction target/mask differs")
        if row.get("node_type") != label["node_type"]:
            raise ValueError("Prediction node type differs")
        for ck in CHECKPOINTS:
            if row[ck] not in (0, 1):
                raise ValueError("Nonbinary prediction")
            p = row[ck + "_positive_probability"]
            if not 0 <= p <= 1:
                raise ValueError("Invalid prediction probability")
    return table


def rules_from_edges(rows, attrs, diagnostics):
    """Frozen F4/attribute-only baselines; unknown is never inferred as false."""
    guard_targets, write_nodes, crossed, unknown_guard, unknown_cross = set(), set(), set(), set(), set()
    context = {}
    diag = {d["edge_key"]: d for d in diagnostics["edges"]}
    if len(diag)!=len(diagnostics["edges"]) or not set(diag)<=set(attrs):
        raise ValueError("Attribute diagnostics contain duplicate or foreign edge keys")
    for row in rows:
        vec = attrs[row["key"]]
        src, dst, rel = row["src"], row["dst"], row["relation"]
        if rel == "WRITES_STATE":
            write_nodes.add(src)
        if rel == "GUARDS" and vec[49]:
            guard_targets.add(dst)
        if vec[72]:
            unknown_guard.add(dst)
        if rel == "DATA_DEF_USE":
            if "21" in diag.get(row["key"],{}).get("unknown_fields",{}):
                unknown_cross.add(dst)
            if vec[21]:
                crossed.add(dst)
        if rel in ("GUARDS", "CALL_SITE", "MOD_APPLY"):
            for node in (src, dst):
                item = context.setdefault(node, set())
                item.add(rel)
                if rel == "GUARDS" and vec[47]:
                    item.add("modifier_guard")
    return dict(
        no_caller_guard=dict(positive=write_nodes - guard_targets - unknown_guard,
                             excluded=unknown_guard, name="state_write_without_caller_identity_GUARDS"),
        cross_call_to_write=dict(positive=write_nodes & crossed,
                                 excluded=(write_nodes & unknown_cross) - crossed,
                                 name="state_write_with_use_path_crosses_ext_call"),
        context=context, write_nodes=write_nodes, guard_targets=guard_targets)


def metadata(source, inputs, task_id):
    task = source["tasks"][task_id]
    labels = scoring.index_rows(read(checked(task["label_view"]))[task_id])
    lexical = scoring.index_rows(read(checked(inputs["node_diagnostic_views"]))[task_id])
    sources = task["record"]["sources"]
    for reference in sources.values():
        checked(reference)
    for key, row in labels.items():
        if row["source_sha256"] != sources[row["file"]]["sha256"]:
            raise ValueError("Frozen label/source identity differs")
        if row["node_type"] == "env_source" and not row["L-audit_include"]:
            continue
        old = lexical.get(key)
        if (old is None or old["source_sha256"] != row["source_sha256"]
                or old["solidifi"] != row["solidifi"]):
            raise ValueError("Lexical diagnostic identity differs")
    attrs_ref = inputs["attrs"][task_id]
    manifest = read(checked(attrs_ref))
    graph = read_graph(checked(manifest["input_provenance"]["graphs"]["F3"]))
    f4 = load_edges(checked(manifest["f4_manifest"]))
    attrs = load_attributes(checked(attrs_ref), graph, f4)
    rows = edge_rows(graph, f4)
    rules = rules_from_edges(rows, attrs, read(checked(manifest["diagnostics"])))
    return labels, lexical, rules


def paired_tables(records, by_profile, view, task_id):
    comparisons, error_rows, changes = [], [], []
    for ck in CHECKPOINTS:
        for profile, baseline in CONTRASTS:
            for metric in stats.METRICS:
                result = stats.paired_comparison(records, task_id, profile, baseline, ck, metric)
                result.update(view=view, analysis_role="primary" if view == "original" else "sensitivity",
                              confirmatory_component=(view == "original" and task_id == "reentrancy_node"
                                  and ck == "best" and metric == "buggy_f1"
                                  and (profile, baseline) in CONTRASTS[:2]),
                              Holm_performed=False)
                comparisons.append(result)
            for fold in range(10):
                left = scoring.index_rows(by_profile[profile][fold])
                right = scoring.index_rows(by_profile[baseline][fold])
                if set(left) != set(right) or any(left[k]["target"] != right[k]["target"] for k in left):
                    raise ValueError("Paired fold identity/target differs")
                a, b = score(left.values(), ck), score(right.values(), ck)
                corrected = introduced = 0
                probability_deltas = []
                for key in left:
                    new, old = left[key], right[key]
                    if new[ck] != old[ck]:
                        good = new[ck] == new["target"]
                        corrected += int(good)
                        introduced += int(not good)
                        changes.append(dict(view=view, task_id=task_id, fold=fold, checkpoint=ck,
                            profile=profile, baseline=baseline, file=key[0], original_graph_node_id=key[1],
                            target=new["target"], previous=old[ck], current=new[ck],
                            corrected=good, previous_probability=old[ck+"_positive_probability"],
                            current_probability=new[ck+"_positive_probability"]))
                    probability_deltas.append(new[ck+"_positive_probability"] - old[ck+"_positive_probability"])
                error_rows.append(dict(view=view, task_id=task_id, fold=fold, checkpoint=ck,
                    profile=profile, baseline=baseline, baseline_errors=b["errors"], errors=a["errors"],
                    relative_error_reduction_pct=relative_error_reduction(b["errors"], a["errors"]),
                    corrected=corrected, introduced=introduced,
                    mean_probability_change=statistics.mean(probability_deltas),
                    mean_absolute_probability_change=statistics.mean(abs(x) for x in probability_deltas),
                    max_absolute_probability_change=max(abs(x) for x in probability_deltas)))
    return comparisons, error_rows, changes


def summarize(root, output):
    root, output = Path(root), Path(output)
    index_path,index = preparation_index(root)
    source = h8.load(checked(index["source_h8_plan"]))
    inputs = index["analysis_inputs"]
    cases = read(checked(inputs["persistent_cases"]))
    definitions = {
        "original": [checked(index["baseline_plan"]), root/"primary/plan.json"],
        "anonymous": [root/"sensitivity/plan.json"],
    }
    fold_rows, pooled_rows, mean_rows, paired, error_rows, changes = [], [], [], [], [], []
    node_cases, persistent, rule_scores, resources, supports, accepted = [], [], [], [], [], []
    labels_by_task = {}
    inventories = {}
    for view, paths in definitions.items():
        by_task = {t: {p: {} for p in PROFILES} for t in ex.TASKS}
        for path in paths:
            plan = ex.load(path, mode="verify")
            if ex.selected_view(plan) != view:
                raise ValueError("Report cannot mix token views")
            for job in plan["jobs"]:
                item = ex.verify_one(path, plan, job)
                task_id, profile, fold = job["task_id"], job["edge_profile"], job["fold"]
                if profile not in PROFILES or fold in by_task[task_id][profile]:
                    raise ValueError("Unexpected or duplicate result")
                rows = read(Path(item["attempt"])/"outer_test_predictions.json")
                by_task[task_id][profile][fold] = rows
                accepted.append(dict(view=view, job_id=ex.job_id(job), plan=ref(path),
                                     receipt=ref(Path(item["attempt"])/"receipt.json")))
                r = item["result"]
                resources.append(dict(view=view, task_id=task_id, profile=profile, fold=fold,
                    best_epoch_1based=r["best_epoch"]+1, last_epoch_1based=r["last_epoch"]+1,
                    seconds=r["seconds"], peak_allocated_mib=r["resources"]["peak_allocated_mib"],
                    peak_reserved_mib=r["resources"]["peak_reserved_mib"],
                    resources=r["resources"], parameter_count=r["parameter_count"],
                    receipt=accepted[-1]["receipt"]))
                supports.append(dict(view=view, task_id=task_id, profile=profile, fold=fold,
                    support=job["support"], gradient_audit=ref(Path(item["attempt"])/"gradient_audit.json")
                    if (Path(item["attempt"])/"gradient_audit.json").is_file() else None))
        inventories[view] = by_task
        for task_id, profiles in by_task.items():
            if any(set(folds) != set(range(10)) for folds in profiles.values()):
                raise ValueError("Incomplete E-A result matrix")
            labels, lexical, rules = metadata(source, inputs, task_id)
            labels_by_task[task_id] = labels
            records, oof = [], {}
            reference_fold_ids = None
            for profile, folds in profiles.items():
                seen, pool, fold_ids = set(), [], {}
                for fold, rows in sorted(folds.items()):
                    # verify_one already checks each fold against its sealed split.
                    table = align_predictions(rows, labels)
                    if seen & set(table):
                        raise ValueError("OOF identity appears in more than one fold")
                    fold_ids[fold] = set(table)
                    seen.update(table)
                    pool.extend(rows)
                    for ck in CHECKPOINTS:
                        counts = score(rows, ck)
                        fold_rows.append(dict(view=view, task_id=task_id, profile=profile, fold=fold,
                                              checkpoint=ck, **counts))
                        records.append(dict(task_id=task_id, task_kind="node", profile=profile,
                            fold=fold, seed=1, checkpoint=ck,
                            metrics={k: counts[k] for k in stats.METRICS}))
                expected_all = {k for k, r in labels.items() if r["L-audit_include"]}
                if seen != expected_all:
                    raise ValueError("OOF does not cover every frozen scored node exactly once")
                if reference_fold_ids is not None and fold_ids != reference_fold_ids:
                    raise ValueError("Profiles use different outer folds")
                reference_fold_ids = fold_ids
                oof[profile] = scoring.index_rows(pool)
                for ck in CHECKPOINTS:
                    for stratum in scoring.STRATA:
                        pooled_rows.append(dict(view=view, task_id=task_id, profile=profile,
                            checkpoint=ck, stratum=stratum, aggregation="pooled_OOF",
                            **score(pool, ck, stratum)))
            for row in stats.aggregate_metrics(records):
                mean_rows.append(dict(view=view, **row))
            comparisons, errs, changed = paired_tables(records, profiles, view, task_id)
            paired.extend(comparisons);error_rows.extend(errs)
            for row in changed:
                label=labels[(row["file"],row["original_graph_node_id"])]
                row.update(source_sha256=label["source_sha256"],source_id=label["source_id"],
                           cohort="SolidiFI" if label["solidifi"] else "nonSolidiFI")
            changes.extend(changed)
            for profile in PROFILES:
                for fold, rows in sorted(profiles[profile].items()):
                    for row in rows:
                        key=scoring.identity(row);label=labels[key]
                        if label["solidifi"]:
                            continue
                        for ck in CHECKPOINTS:
                            if row["target"] or row[ck]:
                                node_cases.append(dict(view=view, task_id=task_id, profile=profile,
                                    fold=fold, checkpoint=ck, file=key[0], original_graph_node_id=key[1],
                                    source_sha256=label["source_sha256"], source_id=label["source_id"],
                                    target=row["target"],prediction=row[ck], probability=row[ck+"_positive_probability"],
                                    classification=("TP" if row[ck] else "FN") if row["target"] else "FP"))
            for case in cases:
                if case["task_id"] != task_id:
                    continue
                key=scoring.identity(case)
                if key not in oof["G3R"]:
                    raise ValueError("Registered persistent case missing")
                label=labels[key]
                if case["source_sha256"] != label["source_sha256"]:
                    raise ValueError("Persistent case source identity differs")
                matching_folds=[fold for fold, rows in profiles["G3R"].items()
                                if key in scoring.index_rows(rows)]
                if matching_folds != [case["fold"]]:
                    raise ValueError("Persistent case fold differs")
                for ck in CHECKPOINTS:
                    node=key[1]
                    persistent.append(dict(view=view, task_id=task_id, checkpoint=ck,
                        file=key[0],original_graph_node_id=node,source_sha256=label["source_sha256"],
                        source_id=label["source_id"],fold=case["fold"],target=label["L-audit_target"],
                        predictions={p:oof[p][key][ck] for p in PROFILES},
                        probabilities={p:oof[p][key][ck+"_positive_probability"] for p in PROFILES},
                        modifier_related=("modifier_guard" in rules["context"].get(node,set())
                            or bool(case.get("facts",{}).get("function_ctx",{}).get("has_modifier"))),
                        statement_call_related="CALL_SITE" in rules["context"].get(node,set()),
                        unprotected_state_write=node in rules["no_caller_guard"]["positive"],
                        frozen_case_type=case.get("case_type"),alignment=case.get("alignment")))
            for rule_name in (scoring.LEXICAL,"no_caller_guard","cross_call_to_write"):
                excluded=set() if rule_name==scoring.LEXICAL else rules[rule_name]["excluded"]
                data=[]
                for key,row in oof["G3R"].items():
                    if key[1] in excluded:
                        continue
                    pred=(lexical[key]["lexical_prediction"] if rule_name==scoring.LEXICAL
                          else int(key[1] in rules[rule_name]["positive"]))
                    data.append(dict(row,best=pred))
                for stratum in scoring.STRATA:
                    all_n=score(oof["G3R"].values(),"best",stratum)["n"]
                    value=score(data,"best",stratum)
                    rule_scores.append(dict(view=view,task_id=task_id,rule=rule_name,stratum=stratum,
                        excluded_unknown=all_n-value["n"],**value,
                        label_policy="L-audit",learning=False,
                        lexical_scope="Frozen original-name audit rule, identical across views; not an anonymous-input predictor"
                            if rule_name==scoring.LEXICAL else None))
    # Same folds and same labels across both views; identity, never row position.
    for task in ex.TASKS:
        for p in PROFILES:
            for fold in range(10):
                a=scoring.index_rows(inventories["original"][task][p][fold])
                b=scoring.index_rows(inventories["anonymous"][task][p][fold])
                if set(a)!=set(b) or any(a[k]["target"]!=b[k]["target"] for k in a):
                    raise ValueError("Cross-view population differs")
    hypotheses={}
    for view in definitions:
        parts=[r for r in paired if r["view"]==view and r["task_id"]=="reentrancy_node"
               and r["checkpoint"]=="best" and r["metric"]=="buggy_f1"
               and (r["profile"],r["baseline"]) in CONTRASTS[:2]]
        if len(parts)!=2:
            raise ValueError("H1 requires both preregistered comparisons")
        hypotheses[view]=dict(H1_raw_p=max(r["p_raw"] for r in parts),
            both_mean_differences_positive=all(r["difference_mean_pp"]>0 for r in parts),
            components=parts, Holm_status="pending_H2_EP9" if view=="original" else "outside_Holm_descriptive")
    if len(accepted)!=160:
        raise ValueError("E-A requires 140 new plus 20 reused fits")
    output.mkdir(parents=True,exist_ok=False)
    tables=dict(folds=fold_rows,pooled=pooled_rows,fold_means=mean_rows,paired=paired,
        error_changes=error_rows,changed_nodes=changes,non_solidifi_nodes=node_cases,
        persistent_cases=persistent,rule_baselines=rule_scores,resources=resources,relation_support=supports)
    for name, rows in tables.items():
        write(output/(name+".json"),rows)
        csv_table(output/(name+".csv"),rows)
    summary=dict(version="edgesem-E-A-analysis-v1",passed=True,new_fits=140,reused_fits=20,
        hypotheses=hypotheses,preparation=ref(index_path),
        accepted=accepted,tables={name:ref(output/(name+".json")) for name in tables},
        analysis_source=ref(Path(__file__)),statistical_source=ref(Path(scoring.__file__)),
        training_launched_by_analysis=False,
        notes=["Original is the sole primary view; anonymous is descriptive sensitivity.",
               "Holm awaits H2. Last is auxiliary; best is the preregistered main checkpoint.",
               "Wilson intervals apply to P/R counts, not F1, and ignore node/file dependence.",
               "Equal-fold means and pooled OOF differ; OOF is not an independent external test.",
               "Graph parameters, aggregation and optimization also differ; score changes are not pure semantic causation.",
               "Original lexical rule is an audit comparator in both views, not a predictor from anonymized inputs."])
    write(output/"summary.json",summary)
    write(output/"summary.seal.json",dict(summary=ref(output/"summary.json")))
    report(root,output,summary,tables)
    return summary
