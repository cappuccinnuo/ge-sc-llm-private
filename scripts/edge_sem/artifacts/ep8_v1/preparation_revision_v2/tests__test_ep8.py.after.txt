def put(path,value):
    path.parent.mkdir(parents=True,exist_ok=True)
    path.write_text(json.dumps(value),encoding="utf8")
    return ref(path)


def registration(tmp_path):
    return put(tmp_path/"registration.json",dict(experiment_stage="E-A",
        primary_endpoint=dict(view="original"),batches={
            "primary":dict(view="original",profiles=["G3R-T","G4R","G4R-P"],holm_member=True),
            "sensitivity":dict(view="anonymous",profiles=list(analysis.PROFILES),holm_member=False)}))


def plan(tmp_path,role):
    profiles=analysis.PROFILES[1:] if role=="primary" else analysis.PROFILES
    return dict(experiment_stage="E-A",analysis_role=role,
        registration=registration(tmp_path),
        token_view="original" if role=="primary" else "anonymous",
        baseline_plan={"path":"historical"},
        prior_main_plan={"path":"primary"} if role=="sensitivity" else None,
        jobs=[dict(task_id=t,fold=f,edge_profile=p) for t in ex.TASKS
              for f in range(10) for p in profiles])


@pytest.mark.parametrize("role",["primary","sensitivity"])
def test_exact_ea_matrices_and_unique_primary(tmp_path,role):
    value=plan(tmp_path,role)
    assert fv.validate_plan_view(value)==value["token_view"]


@pytest.mark.parametrize("damage",["view","role","duplicate","missing","profile","baseline","stage"])
def test_ea_wrong_view_or_matrix_refused(tmp_path,damage):
    value=plan(tmp_path,"primary")
    if damage=="view":value["token_view"]="anonymous"
    elif damage=="role":value["analysis_role"]="sensitivity"
    elif damage=="duplicate":value["jobs"][-1]=value["jobs"][0]
    elif damage=="missing":value["jobs"].pop()
    elif damage=="profile":value["jobs"][0]["edge_profile"]="G4R-A"
    elif damage=="baseline":value["baseline_plan"]=None
    else:value["experiment_stage"]="E-B"
    with pytest.raises(ValueError):fv.validate_plan_view(value)


def test_anonymous_sensitivity_cannot_enter_holm(tmp_path):
    value=plan(tmp_path,"sensitivity")
    path=Path(value["registration"]["path"])
    data=json.loads(path.read_text())
    data["batches"]["sensitivity"]["holm_member"]=True
    value["registration"]=put(path,data)
    with pytest.raises(ValueError):fv.validate_plan_view(value)

    path=tmp_path/"decisions.json"

    with pytest.raises(ValueError):p2.require_decisions(path,"anonymous","E-A")


def test_sensitivity_requires_sealed_completed_primary(tmp_path,monkeypatch):
    # Only EdgeSem functions are replaced; frozen DFG/NodeSem modules are untouched.
    value=plan(tmp_path,"sensitivity")
    value.update(scope="formal")
    prior=dict(experiment_stage="E-A",analysis_role="primary",registration=value["registration"])
    previous=tmp_path/"primary/plan.json";value["prior_main_plan"]=put(previous,prior)
    path=tmp_path/"sensitivity/plan.json";put(path,value)
    baseline=tmp_path/"baseline.json";value["baseline_plan"]=put(baseline,{})
    decisions=tmp_path/"decisions.json";put(decisions,{})
    approval=dict(plan=ref(path),scope="formal",execution_authorized=True,user_request="test",
        jobs=[ex.job_id(dict(j,seed=1)) for j in value["jobs"]],
        decisions=ref(decisions),analysis_role="sensitivity",baseline_plan=value["baseline_plan"],
        registration_commit="test")
    for j in value["jobs"]:j["seed"]=1
    target=path.parent/"execution_authorization.json"
    put(target,approval);put(target.with_suffix(".seal.json"),dict(authorization=ref(target)))
    monkeypatch.setattr(p2,"require_decisions",lambda *a:None)
    monkeypatch.setattr(p2,"require_registration_commit",lambda *a:None)
    with pytest.raises(PermissionError,match="Primary batch must complete first"):
        ex.authorized(path,value)
    complete=previous.parent/"complete.json"
    put(complete,dict(plan=value["prior_main_plan"],passed=True,receipts=[{}]*59))
    with pytest.raises(PermissionError,match="completion differs"):ex.authorized(path,value)
    put(complete,dict(plan=value["prior_main_plan"],passed=True,receipts=[{}]*60))
    put(complete.with_suffix(".seal.json"),dict(complete={"tampered":True}))
    with pytest.raises(ValueError,match="completion seal differs"):ex.authorized(path,value)


def examples():
    labels={}
    rows=[]
    for n,(file,target,pred) in enumerate([("buggy_1.sol",1,1),("buggy_1.sol",0,1),
                                          ("real.sol",1,0),("real.sol",0,0)]):
        row=dict(file=file,original_graph_node_id=n,node_type="expression_statement",
            target=target,best=pred,last=pred,best_positive_probability=.7 if pred else .2,
            last_positive_probability=.7 if pred else .2)
        rows.append(row)
        labels[(file,n)]={
            "L-audit_include":True,"L-audit_target":target,"node_type":"expression_statement"}
    return rows,labels


def test_scoring_reuses_frozen_counts_and_cohorts():
    rows,_=examples()
    all_values=analysis.score(rows)
    assert all_values["tn_fp_fn_tp"]==[1,1,1,1]
    assert all_values["buggy_f1"]==50 and all_values["errors"]==2
    assert analysis.score(rows,stratum="SolidiFI")["tn_fp_fn_tp"]==[0,1,0,1]
    assert analysis.score(rows,stratum="nonSolidiFI")["tn_fp_fn_tp"]==[1,0,1,0]
    assert analysis.score([])["precision_wilson_95"] is None
    assert analysis.relative_error_reduction(0,0) is None
    assert analysis.relative_error_reduction(10,8)==20
    assert analysis.relative_error_reduction(10,12)==-20


def test_prediction_alignment_is_by_identity_not_row_number():
    rows,labels=examples()
    a=analysis.align_predictions(rows,labels)
    b=analysis.align_predictions(list(reversed(rows)),labels,set(a))
    assert a==b
    with pytest.raises(ValueError):analysis.align_predictions(rows+[rows[0]],labels)
    bad=copy.deepcopy(rows);bad[0]["target"]=0
    with pytest.raises(ValueError):analysis.align_predictions(bad,labels)
    with pytest.raises(ValueError):analysis.align_predictions(rows[:-1],labels,set(a))


def test_unknown_guard_is_not_absence_and_only_exact_edge_sink_is_used():
    rows=[
        dict(key="w1",src=1,dst=9,relation="WRITES_STATE"),
        dict(key="w2",src=2,dst=9,relation="WRITES_STATE"),
        dict(key="g",src=3,dst=1,relation="GUARDS"),
        dict(key="u",src=4,dst=2,relation="next_line"),
        dict(key="d",src=5,dst=2,relation="DATA_DEF_USE"),
    ]
    attrs={r["key"]:np.zeros(74,dtype=np.uint8) for r in rows}
    attrs["g"][49]=1;attrs["u"][72]=1;attrs["d"][21]=1
    diagnostics=dict(edges=[dict(edge_key=r["key"],unknown_fields={}) for r in rows])
    result=analysis.rules_from_edges(rows,attrs,diagnostics)
    assert result["no_caller_guard"]["positive"]==set()
    assert result["no_caller_guard"]["excluded"]=={2}
    assert result["cross_call_to_write"]["positive"]=={2}
    attrs["d"][21]=0
    diagnostics["edges"][-1]["unknown_fields"]={"21":["cfg unavailable"]}
    result=analysis.rules_from_edges(rows,attrs,diagnostics)
    assert result["cross_call_to_write"]["positive"]==set()
    assert result["cross_call_to_write"]["excluded"]=={2}


def test_paired_zero_folds_use_frozen_zsplit():
    rows,_=examples();records=[];profiles={}
    for p in analysis.PROFILES:
        profiles[p]={f:rows for f in range(10)}
        for f in range(10):
            for ck in analysis.CHECKPOINTS:
                records.append(dict(task_id="reentrancy_node",task_kind="node",profile=p,fold=f,seed=1,
                    checkpoint=ck,metrics={k:analysis.score(rows,ck)[k] for k in stats.METRICS}))
    comparisons,errors,changes=analysis.paired_tables(records,profiles,"anonymous","reentrancy_node")
    assert len(comparisons)==24 and len(errors)==60 and not changes
    assert all(r["tied"]==10 and r["p_raw"]==1 and not r["confirmatory_component"] for r in comparisons)
    assert all(r["relative_error_reduction_pct"]==0 for r in errors)


def test_protocol_additions_do_not_change_numeric_training_functions():
    import ast
    before=Path(__file__).resolve().parents[1]/"artifacts/ep8_v1/baseline/scripts__edge_sem__execution.py.txt"
    if not before.is_file():pytest.skip("Local EP8 baseline snapshot absent")
    current=Path(ex.__file__)
    old={n.name:ast.dump(n) for n in ast.parse(before.read_text()).body if isinstance(n,ast.FunctionDef)}
    new={n.name:ast.dump(n) for n in ast.parse(current.read_text()).body if isinstance(n,ast.FunctionDef)}
    for name in ("assemble_model","construct","binding_parts","fit_one","_verify_result"):
        assert old[name]==new[name]

from scripts.DFG.dfg_v2.tests.test_model import hcg,augment


def test_preparation_view_installation_matches_fresh_model_and_rejects_training(hcg,tmp_path):
    import hashlib
    import networkx as nx
    import torch
    from scripts.DFG.dfg_v2 import AUX_NODE_TYPES
    from scripts.DFG.dfg_v2.benchmark_v2 import spelling_features as sf
    from scripts.DFG.dfg_v2.benchmark_v2.feature_view import graph_fingerprint,vector_sha
    from scripts.edge_sem.model import make_edge_model
    from scripts.edge_sem.recon.ep8_run import install_untrained_view,topology_equal
    graph=augment(hcg,"G3R")
    path=tmp_path/"graph.gpickle";nx.write_gpickle(graph,path)
    rows=[];values=[];sources={}
    for n,a in graph.nodes(data=True):
        if a["node_type"] in AUX_NODE_TYPES:continue
        sources[a["source_file"]]={"sha256":"synthetic-source"}
        vector=np.full(512,len(rows)+1,dtype=np.float32);values.append(vector)
        rows.append(dict(original_graph_node_id=n,file=a["source_file"],node_type=a["node_type"],
            source_sha256="synthetic-source",code_sha256=hashlib.sha256(a["code"].encode()).hexdigest(),
            original_vector_sha256=vector_sha(a["token"]),anonymous_vector_sha256=vector_sha(vector)))
    np.savez(tmp_path/"features.npz",features=np.stack(values))
    manifest=put(tmp_path/"view.json",dict(version=sf.VERSION,view="anonymous",
        profiles={"G3R":{"graph_fingerprint":graph_fingerprint(graph)}},rows=rows,sources=sources,
        data=ref(tmp_path/"features.npz"),feature_shape=[len(rows),512],vector_sha_field="anonymous_vector_sha256"))
    original=make_edge_model("node",graph,"G3R-T",hidden_size=16,layers=1,heads=2)
    before=graph_fingerprint(original.nx_graph)
    job=dict(edge_profile="G3R-T",seeds=original.edge_seeds,token_view="anonymous",
             graph=ref(path),feature_view=manifest)
    fast=install_untrained_view(original,job)
    full=make_edge_model("node",sf.assemble(graph,manifest),"G3R-T",hidden_size=16,layers=1,heads=2)
    topology_equal(fast,full)
    assert graph_fingerprint(original.nx_graph)==before
    assert graph_fingerprint(fast.nx_graph)==graph_fingerprint(full.nx_graph)
    assert all(torch.equal(v,full.state_dict()[k]) for k,v in fast.state_dict().items())
    for typ in fast.symmetrical_global_graph.ntypes:
        assert torch.equal(fast.symmetrical_global_graph.nodes[typ].data["inp"],
                           full.symmetrical_global_graph.nodes[typ].data["inp"])
    fast.eval();full.eval()
    with torch.no_grad():assert torch.equal(fast(),full())
    original.edge_optimizer_steps=1
    with pytest.raises(ValueError,match="untrained"):install_untrained_view(original,job)
    original.edge_optimizer_steps=0
    next(original.parameters()).grad=torch.zeros_like(next(original.parameters()))
    with pytest.raises(ValueError,match="untrained"):install_untrained_view(original,job)
