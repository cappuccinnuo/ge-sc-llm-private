"""Recorded shell-level EP9 follow-up; delegates all execution to ep9_run."""
import json
import os
from pathlib import Path
import re
import subprocess
import time
import traceback
from scripts.edge_sem.recon import ep9_run as driver
from scripts.edge_sem.io import ROOT, read, ref, checked, write

root=ROOT/"scripts/edge_sem/artifacts/ep9_v1"
folder=root/"recovery_sampling_v2"
watch=read(folder/"followup_process.json")
try:
    pid=watch["cpu_tests_pid"]
    while Path("/proc/%d" % pid).exists():
        cmd=Path("/proc/%d/cmdline" % pid).read_bytes()
        if b"scripts.edge_sem.recon.ep9_run" not in cmd or b"recovery_sampling_v2" not in cmd:
            break
        state=Path("/proc/%d/stat" % pid).read_text().split(") ",1)[1]
        if state.startswith("Z "):
            break
        time.sleep(15)
    if (folder/"cpu_tests/failure.json").exists():
        raise RuntimeError("CPU regression failed; retained, no automatic retry")
    tests=driver.sealed_json(folder/"cpu_tests/results.json","results")
    if any(r["returncode"] for r in tests["results"]):
        raise RuntimeError("CPU regressions did not pass")
    print("EP9 recovery CPU suites passed",flush=True)
    revision=driver.seal_sampling_recovery(root,folder)
    counts={}
    for row in tests["results"]:
        matches=re.findall(r"(\d+) passed",checked(row["log"]).read_text())
        counts[row["suite"]]=int(matches[-1]) if matches else None
    text=("\n\n## EP9：监控修复验收通过并恢复原批次（2026-11-13）\n\n"
          "recovery_sampling_v1 的完整回归保留为失败现场：409项通过，唯一失败是旧EP8测试要求监控包装函数也完全不变。"
          "按本轮已授权的监控修复更新该测试边界，训练块、模型构造与所有旧结果校验继续严格比对；修正后相关31项定向检查通过。"
          "自动审批曾将第二个CPU验收目录误判为另需授权，补充用户本轮原文和只读证据后已放行，未绕过审批。\n\n"
          "最终采用 recovery_sampling_v2。完整CPU回归：" + json.dumps(counts,ensure_ascii=False) + "；"
          "4份成功fit的收据、权重、预测及指标已复核并复用；冻结源码、旧产物和文档前缀核验通过。"
          "独立 runtime_revision.json / seal 绑定新的监控实现与原120个任务，只有失败的第5项登记 attempt_002，从同一零步初值重跑。"
          "本地提交后按原名60次→匿名60次执行；不重跑已成功项，不修改模型、初值、图、标签、折或P2训练规则。"
          "此处是恢复记录，完整完成以 ep9_v1/pipeline_complete.json、EP9_REPORT.md 和 EDGESEM_REPORT.md 为准。\n\n"
          "实际命令：python -B -m scripts.edge_sem.recon.ep9_run tests --recovery scripts/edge_sem/artifacts/ep9_v1/recovery_sampling_v2；"
          "seal-recovery 同 --recovery；run 同 --recovery。最终实验仍在显存充足的服务器另建批次，本轮仅本机EP9。\n")
    offsets=[]
    for name in ("HANDOVER.md","docs/training_changes_2026-10-07.md","scripts/edge_sem/PROGRESS.md"):
        p=ROOT/name
        old=p.read_bytes()
        with p.open("a",encoding="utf8") as stream:stream.write(text)
        if p.read_bytes()[:len(old)]!=old:
            raise ValueError("Document prefix changed")
        offsets.append(str(p)+":"+str(len(old)))
    subprocess.run([os.sys.executable,"-B","-m","scripts.node_sem.doc_check","--expect-cjk"]+offsets,
                   check=True,cwd=str(ROOT))
    write(folder/"documentation_accepted.json",dict(counts=counts,revision=revision,offsets=offsets))
    driver.commit_delivery(root,"EP9 repair transient memory sampling and seal explicit recovery")
    commit=subprocess.check_output(["git","rev-parse","HEAD"],cwd=str(ROOT),universal_newlines=True).strip()
    write(folder/"recovery_commit.json",dict(commit=commit,revision=revision))
    print("EP9 resume original 60 then anonymous 60; reusing four accepted fits",flush=True)
    driver.run(root,recovery=folder)
    write(folder/"followup_complete.json",dict(passed=True,pipeline=ref(root/"pipeline_complete.json")))
except Exception as error:
    write(folder/"followup_failure.json",dict(error=repr(error),traceback=traceback.format_exc(),
          automatic_retry=False))
    raise
